    30→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
    31→  let score = 0
    32→  if (password.length >= 6) score++
    33→  if (password.length >= 8) score++
    34→  if (/[A-Z]/.test(password)) score++
    35→  if (/[0-9]/.test(password)) score++
    36→  if (/[^A-Za-z0-9]/.test(password)) score++
    37→
    38→  if (score <= 1) return { score: 20, label: "Muy débil", color: "bg-red-500" }
    39→  if (score === 2) return { score: 40, label: "Débil", color: "bg-orange-500" }
    40→  if (score === 3) return { score: 60, label: "Regular", color: "bg-yellow-500" }
    41→  if (score === 4) return { score: 80, label: "Fuerte", color: "bg-green-500" }
    42→  return { score: 100, label: "Muy fuerte", color: "bg-emerald-500" }
    43→}
