   422→  // ─── Like Post ──────────────────────────────────────────────────────────────

   424→  const handleLikePost = async (postId: string) => {

   428→      const res = await authFetch("/api/wall/like", {
   429→        method: "POST",
   430→        headers: { "Content-Type": "application/json" },

   433→      const d = await res.json()

   436→        // Update post like count
   437→        setWallPosts(prev => prev.map(p => {
   438→          if (p.id !== postId) return p
   439→          const countDelta = d.data.liked ? 1 : -1
   440→          return { ...p, _count: { ...(p._count ?? { likes: 0, comments: 0 }), likes: Math.max(0, (p._count?.likes ?? 0) + countDelta) } }
   441→        }))
   442→      }
   443→    } catch {
   444→      setLikedPosts(prev => ({ ...prev, [postId]: wasLiked }))
   445→    }
   446→  }

   495→        toast.error(d.error || "Error al comentar")
   496→      }
   497→    } catch {
   498→      toast.error("Error al comentar")

   502→  }

   562→  const formatPrice = (p: number) => new Intl.NumberFormat("es-NI", { style: "currency", currency: "NIO" }).format(p)
   563→
   564→  const timeAgo = (dateStr: string) => {
   565→    const diff = Date.now() - new Date(dateStr).getTime()
   566→    const mins = Math.floor(diff / 60000)
   567→    if (mins < 1) return "Ahora"
   568→    if (mins < 60) return `hace ${mins}m`
   569→    const hrs = Math.floor(mins / 60)
   570→    if (hrs < 24) return `hace ${hrs}h`
   571→    const days = Math.floor(hrs / 24)
   572→    if (days < 7) return `hace ${days}d`
   573→    return new Date(dateStr).toLocaleDateString("es-NI", { day: "numeric", month: "short" })
   574→  }
