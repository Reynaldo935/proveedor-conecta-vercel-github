   113→// ─── Animated Counter Component ──────────────────────────────────────────────
   114→function AnimatedCounter({ target, suffix = "" }: { target: number; suffix?: string }) {
   115→  const [count, setCount] = useState(0)
   116→  const hasAnimated = useRef(false)
   117→
   118→  useEffect(() => {
   119→    if (hasAnimated.current) return
   120→    hasAnimated.current = true
   121→    const duration = 2000
   122→    const steps = 60
   123→    const increment = target / steps
   124→    let current = 0
   125→    const timer = setInterval(() => {
   126→      current += increment
   127→      if (current >= target) {
   128→        setCount(target)
   129→        clearInterval(timer)
   130→      } else {
   131→        setCount(Math.floor(current))
   132→      }
   133→    }, duration / steps)
   134→    return () => clearInterval(timer)
   135→  }, [target])
   136→
   137→  return (
   138→    <span suppressHydrationWarning>
   139→      {count.toLocaleString()}{suffix}
   140→    </span>
   141→  )
   142→}

   206→export function HomeFeed() {
   207→  const { navigate, selectedCategory, setSelectedCategory, setSearchQuery, selectedLocation, setSelectedLocation } = useAppStore()
   208→  const { user, isAuthenticated } = useAuthStore()
   209→
   210→  // ─── State ──────────────────────────────────────────────────────────────
   211→  const [products, setProducts] = useState<Product[]>([])
   212→  const [loading, setLoading] = useState(true)
   213→  const [cursor, setCursor] = useState<string | null>(null)
   214→  const [hasMore, setHasMore] = useState(true)
   215→  const [searchInput, setSearchInput] = useState("")
   216→  const [showScrollTop, setShowScrollTop] = useState(false)
   217→  const [likedProducts, setLikedProducts] = useState<Set<string>>(new Set())
   218→  const [savedProducts, setSavedProducts] = useState<Set<string>>(new Set())
   219→  const [messagingProductId, setMessagingProductId] = useState<string | null>(null)
   220→
   221→  const observerRef = useRef<IntersectionObserver | null>(null)
   222→  const loadMoreRef = useRef<HTMLDivElement | null>(null)
   223→  const loadingRef = useRef(false)
   224→
   225→  // ─── Load Products ──────────────────────────────────────────────────────
   226→  const loadProducts = useCallback(async (reset = false) => {
   227→    if (!reset && !hasMore) return
   228→    if (!reset && loadingRef.current) return
   229→    loadingRef.current = true
   230→    setLoading(true)
   231→    try {
   232→      const params = new URLSearchParams()
   233→      if (!reset && cursor) params.set("cursor", cursor)
   234→      if (selectedCategory) params.set("category", selectedCategory)
   235→      params.set("limit", "20")
   236→
   237→      const res = await fetch(`/api/products?${params}`)
   238→      if (!res.ok) return
   239→      let data: { success?: boolean; data?: Product[]; nextCursor?: string | null }
   240→      try {
   241→        data = await res.json()
   242→      } catch {
   243→        return // Non-JSON response — bail out gracefully
   244→      }
   245→
   246→      if (data.success && Array.isArray(data.data)) {
   247→        const newProducts = data.data
   248→        if (reset) {
   249→          // Deduplicate by ID in case API returns duplicates
   250→          const seen = new Set<string>()
   251→          const unique = newProducts.filter(p => {
   252→            if (seen.has(p.id)) return false
   253→            seen.add(p.id)
   254→            return true
   255→          })
   256→          setProducts(unique)
   257→        } else {
   258→          setProducts(prev => {
   259→            const existing = new Set(prev.map(p => p.id))
   260→            const filtered = newProducts.filter(p => !existing.has(p.id))
   261→            return [...prev, ...filtered]
   262→          })
   263→        }
   264→        setCursor(data.nextCursor ?? null)
   265→        setHasMore(!!data.nextCursor && newProducts.length > 0)
   266→      }
   267→    } catch {
   268→      // Network or parsing error — silently handle so the UI
   269→      // shows the empty state instead of crashing
   270→    } finally {
   271→      loadingRef.current = false
   272→      setLoading(false)
   273→    }
   274→  }, [cursor, selectedCategory, hasMore])
   275→
   276→  useEffect(() => {
   277→    setProducts([])
   278→    setCursor(null)
   279→    setHasMore(true)
   280→    loadProducts(true)
   281→  }, [selectedCategory])
   282→
   283→  // ─── Infinite Scroll ────────────────────────────────────────────────────
   284→  useEffect(() => {
   285→    if (observerRef.current) observerRef.current.disconnect()
   286→
   287→    observerRef.current = new IntersectionObserver(
   288→      (entries) => {
   289→        if (entries[0].isIntersecting && hasMore && !loadingRef.current) {
   290→          loadProducts()
   291→        }
   292→      },
   293→      { threshold: 0.1 }
   294→    )
   295→
   296→    if (loadMoreRef.current) {
   297→      observerRef.current.observe(loadMoreRef.current)
   298→    }
   299→
   300→    return () => observerRef.current?.disconnect()
   301→  }, [hasMore, loading, loadProducts])
   302→
   303→  // ─── Scroll to top visibility ────────────────────────────────────────────
   304→  useEffect(() => {
   305→    const handleScroll = () => {
   306→      setShowScrollTop(window.scrollY > 600)
   307→    }
   308→    window.addEventListener("scroll", handleScroll, { passive: true })
   309→    return () => window.removeEventListener("scroll", handleScroll)
   310→  }, [])
   311→
   312→  // ─── Toggle Like (uses authFetch) ───────────────────────────────────────
   313→  const toggleLike = async (e: React.MouseEvent, productId: string) => {
   314→    e.stopPropagation()
   315→    if (!isAuthenticated) {
   316→      toast.error("Inicia sesión para dar like")
   317→      navigate("login")
   318→      return
   319→    }
   320→    try {
   321→      const res = await authFetch("/api/likes", {
   322→        method: "POST",
   323→        headers: { "Content-Type": "application/json" },
   324→        body: JSON.stringify({ productId }),
   325→      })
   326→      const data = await res.json()
   327→      if (data.success) {
   328→        const isLiked = data.data.liked
   329→        setLikedProducts(prev => {
   330→          const next = new Set(prev)
   331→          if (isLiked) next.add(productId)
   332→          else next.delete(productId)
   333→          return next
   334→        })
   335→        setProducts(prev =>
   336→          prev.map(p =>
   337→            p.id === productId
   338→              ? { ...p, likeCount: isLiked ? p.likeCount + 1 : Math.max(0, p.likeCount - 1) }
   339→              : p
   340→          )
   341→        )
   342→      }
   343→    } catch {
   344→      toast.error("Error al dar like")
   345→    }
   346→  }
   347→
   348→  // ─── Toggle Save (uses authFetch) ───────────────────────────────────────
   349→  const toggleSave = async (e: React.MouseEvent, productId: string) => {
   350→    e.stopPropagation()
   351→    if (!isAuthenticated) {
   352→      toast.error("Inicia sesión para guardar")
   353→      navigate("login")
   354→      return
   355→    }
   356→    try {
   357→      const res = await authFetch("/api/saved", {
   358→        method: "POST",
   359→        headers: { "Content-Type": "application/json" },
   360→        body: JSON.stringify({ productId }),

   362→      const data = await res.json()

   364→        setSavedProducts(prev => {
   365→          const next = new Set(prev)
   366→          if (next.has(productId)) {
   367→            next.delete(productId)
   368→            toast.success("Eliminado de guardados")
