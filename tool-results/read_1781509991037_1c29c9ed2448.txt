     1→"use client"
     2→
     3→import { useState, useEffect, useRef } from "react"
     4→import { useAppStore } from "@/store/app-store"
     5→import { useAuthStore } from "@/store/auth-store"
     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
     7→import { Button } from "@/components/ui/button"
     8→import { Input } from "@/components/ui/input"
     9→import { Label } from "@/components/ui/label"
    10→import { Textarea } from "@/components/ui/textarea"
    11→import {
    12→  Select,
    13→  SelectContent,
    14→  SelectItem,
    15→  SelectTrigger,
    16→  SelectValue,
    17→} from "@/components/ui/select"
    18→import { Badge } from "@/components/ui/badge"
    19→import { Progress } from "@/components/ui/progress"
    20→import { toast } from "sonner"
    21→import { PRODUCT_CATEGORIES } from "@/lib/validators"
    22→import { authFetch } from "@/lib/client-auth"
    23→import {
    24→  Upload,
    25→  X,
    26→  ChevronLeft,
    27→  ChevronRight,
    28→  ImagePlus,
    29→  DollarSign,
    30→  Tag,
    31→  Camera,
    32→  Check,
    33→  Loader2,
    34→  Package,
    35→  Plus,
    36→  Trash2,
    37→} from "lucide-react"
    38→import { motion, AnimatePresence } from "framer-motion"
    39→
    40→interface QuantityDiscountRule {
    41→  minQty: number
    42→  discountPercent: number
    43→}
    44→
    45→interface ProductForm {
    46→  title: string
    47→  description: string
    48→  price: string
    49→  discountPrice: string
    50→  discountPercent: string
    51→  category: string
    52→  tags: string
    53→  images: string[]
    54→  videoUrl: string
    55→  quantity: string
    56→  discountStart: string
    57→  discountEnd: string
    58→  quantityDiscounts: QuantityDiscountRule[]
    59→}
    60→
    61→const STEP_INFO = [
    62→  { label: "Fotos", icon: Camera },
    63→  { label: "Detalles", icon: Tag },
    64→  { label: "Descuento", icon: DollarSign },
    65→]
    66→
    67→export function SellProductForm({ editMode = false }: { editMode?: boolean }) {
    68→  const { navigate, editProductId } = useAppStore()
    69→  const { user } = useAuthStore()
    70→  const [step, setStep] = useState(1)
    71→  const [loading, setLoading] = useState(false)
    72→  const [uploading, setUploading] = useState(false)
    73→  const [uploadProgress, setUploadProgress] = useState(0)
    74→  const [dragActive, setDragActive] = useState(false)
    75→  const [form, setForm] = useState<ProductForm>({
    76→    title: "",
    77→    description: "",
    78→    price: "",
    79→    discountPrice: "",
    80→    discountPercent: "",
    81→    category: "",
    82→    tags: "",
    83→    images: [],
    84→    videoUrl: "",
    85→    quantity: "1",
    86→    discountStart: "",
    87→    discountEnd: "",
    88→    quantityDiscounts: [],
    89→  })
    90→  const [errors, setErrors] = useState<Record<string, string>>({})
    91→  const fileInputRef = useRef<HTMLInputElement>(null)
    92→
    93→  useEffect(() => {
    94→    if (editMode && editProductId) {
    95→      authFetch(`/api/products/${editProductId}`)
    96→        .then((r) => r.json())
    97→        .then((d) => {
    98→          if (d.success) {
    99→            const p = d.data
   100→            setForm({
   101→              title: p.title,
   102→              description: p.description,
   103→              price: String(p.price),
   104→              discountPrice: p.discountPrice ? String(p.discountPrice) : "",
   105→              discountPercent: p.discountPercent
   106→                ? String(p.discountPercent)
   107→                : "",
   108→              category: p.category,
   109→              tags: p.tags,
   110→              images: p.images || [],
   111→              videoUrl: p.videoUrl,
   112→              quantity: String(p.quantity),
   113→              discountStart: p.discountStart
   114→                ? p.discountStart.split("T")[0]
   115→                : "",
   116→              discountEnd: p.discountEnd
   117→                ? p.discountEnd.split("T")[0]
   118→                : "",
   119→              quantityDiscounts: p.quantityDiscounts
   120→                ? p.quantityDiscounts.map((qd: { minQty: number; discountPercent: number }) => ({
   121→                    minQty: qd.minQty,
   122→                    discountPercent: qd.discountPercent,
   123→                  }))
   124→                : [],
   125→            })
   126→          }
   127→        })
   128→        .catch(() => toast.error("Error al cargar producto"))
   129→    }
   130→  }, [editMode, editProductId])
   131→
   132→  const handleUpload = async (files: FileList) => {
   133→    if (form.images.length + files.length > 5) {
   134→      toast.error("Máximo 5 fotos")
   135→      return
   136→    }
   137→    setUploading(true)
   138→    setUploadProgress(0)
   139→    const fd = new FormData()
   140→    Array.from(files).forEach((f) => fd.append("files", f))
   141→    fd.append("subfolder", "products")
   142→    try {
   143→      // Simulate progress for UX
   144→      const progressInterval = setInterval(() => {
   145→        setUploadProgress((prev) => Math.min(prev + 15, 90))
   146→      }, 200)
   147→
   148→      const res = await authFetch("/api/upload", { method: "POST", body: fd })
   149→      clearInterval(progressInterval)
   150→      setUploadProgress(100)
   151→
   152→      const d = await res.json()
   153→      if (d.success) {
   154→        setForm((f) => ({ ...f, images: [...f.images, ...d.data] }))
   155→        toast.success("Imágenes subidas")
   156→        if (fileInputRef.current) fileInputRef.current.value = ""
   157→      } else toast.error(d.error || "Error al subir imágenes")
   158→    } catch {
   159→      toast.error("Error al subir")
   160→    } finally {
   161→      setTimeout(() => {
   162→        setUploading(false)
   163→        setUploadProgress(0)
   164→      }, 500)
   165→    }
   166→  }
   167→
   168→  const removeImage = (idx: number) =>
   169→    setForm((f) => ({ ...f, images: f.images.filter((_, i) => i !== idx) }))
   170→
   171→  const handleDrag = (e: React.DragEvent) => {
   172→    e.preventDefault()
   173→    e.stopPropagation()
   174→    if (e.type === "dragenter" || e.type === "dragover") {
   175→      setDragActive(true)
   176→    } else if (e.type === "dragleave") {
   177→      setDragActive(false)
   178→    }
   179→  }
   180→
   181→  const handleDrop = (e: React.DragEvent) => {
   182→    e.preventDefault()
   183→    e.stopPropagation()
   184→    setDragActive(false)
   185→    if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
   186→      handleUpload(e.dataTransfer.files)
   187→    }
   188→  }
   189→
   190→  const validateStep = (s: number): boolean => {
   191→    const errs: Record<string, string> = {}
   192→    if (s === 1) {
   193→      if (form.images.length === 0) errs.images = "Sube al menos 1 foto"
   194→    } else if (s === 2) {
   195→      if (!form.title.trim()) errs.title = "Título requerido"
   196→      if (!form.price || parseFloat(form.price) <= 0)
   197→        errs.price = "Precio debe ser mayor a 0"
   198→      if (!form.category) errs.category = "Categoría requerida"
   199→      if (form.description.trim().length < 10)
   200→        errs.description = "Descripción debe tener al menos 10 caracteres"
   201→    }
   202→    setErrors(errs)
   203→    return Object.keys(errs).length === 0
   204→  }
   205→
   206→  const handleSubmit = async () => {
   207→    if (!validateStep(2)) return
   208→    setLoading(true)
   209→    try {
   210→      const url = editMode
   211→        ? `/api/products/${editProductId}`
   212→        : "/api/products"
   213→      const method = editMode ? "PUT" : "POST"
   214→      const res = await authFetch(url, {
   215→        method,
   216→        headers: { "Content-Type": "application/json" },
   217→        body: JSON.stringify({
   218→          ...form,
   219→          price: parseFloat(form.price),
   220→          quantity: parseInt(form.quantity, 10),
   221→          discountPrice: form.discountPrice ? parseFloat(form.discountPrice) : null,
   222→          discountPercent: form.discountPercent ? parseFloat(form.discountPercent) : null,
   223→        }),
   224→      })
   225→      const d = await res.json()
   226→      if (d.success) {
   227→        toast.success(editMode ? "Producto actualizado" : "Producto publicado")
   228→        navigate("my-products")
   229→      } else toast.error(d.error || "Error al guardar producto")
   230→    } catch {
   231→      toast.error("Error al guardar")
   232→    } finally {
   233→      setLoading(false)
   234→    }
   235→  }
   236→
   237→  const stepProgress = (step / 3) * 100
   238→
   239→  return (
   240→    <div className="max-w-2xl mx-auto space-y-6">
   241→      <Button variant="ghost" onClick={() => navigate("home")}>
   242→        <ChevronLeft className="h-4 w-4 mr-1" /> Volver
   243→      </Button>
   244→
   245→      <Card>
   246→        <CardHeader>
   247→          <CardTitle className="font-[family-name:var(--font-poppins)]">
   248→            {editMode ? "Editar Producto" : "Vender Producto"}
   249→          </CardTitle>
   250→
   251→          {/* Step progress */}
   252→          <div className="mt-3">
   253→            <Progress value={stepProgress} className="h-2" />
   254→            <div className="flex justify-between mt-2">
   255→              {STEP_INFO.map((info, i) => {
   256→                const stepNum = i + 1
   257→                const isActive = step === stepNum
   258→                const isCompleted = step > stepNum
   259→                return (
   260→                  <div
   261→                    key={i}
   262→                    className={`flex items-center gap-1.5 text-xs cursor-pointer transition-colors ${
   263→                      isCompleted
   264→                        ? "text-primary"
   265→                        : isActive
   266→                          ? "text-primary font-semibold"
   267→                          : "text-muted-foreground"
   268→                    }`}
   269→                    onClick={() => {
   270→                      if (isCompleted) setStep(stepNum)
   271→                    }}
   272→                  >
   273→                    <div
   274→                      className={`h-6 w-6 rounded-full flex items-center justify-center text-[10px] font-bold transition-colors ${
   275→                        isCompleted
   276→                          ? "bg-primary text-primary-foreground"
   277→                          : isActive
   278→                            ? "bg-primary text-primary-foreground"
   279→                            : "bg-muted text-muted-foreground"
   280→                      }`}
   281→                    >
   282→                      {isCompleted ? (
   283→                        <Check className="h-3 w-3" />
   284→                      ) : (
   285→                        stepNum
   286→                      )}
   287→                    </div>
   288→                    <span className="hidden sm:inline">{info.label}</span>
   289→                  </div>
   290→                )
   291→              })}
   292→            </div>
   293→          </div>
   294→        </CardHeader>
   295→
   296→        <CardContent>
   297→          <AnimatePresence mode="wait">
   298→            {/* Step 1: Photos */}
   299→            {step === 1 && (
   300→              <motion.div
   301→                key="step1"
   302→                initial={{ opacity: 0, x: 20 }}
   303→                animate={{ opacity: 1, x: 0 }}
   304→                exit={{ opacity: 0, x: -20 }}
   305→                transition={{ duration: 0.2 }}
   306→                className="space-y-4"
   307→              >
   308→                {/* Drop zone */}
   309→                <div
   310→                  className={`border-2 border-dashed rounded-xl p-8 text-center transition-all ${
   311→                    dragActive
   312→                      ? "border-primary bg-primary/5 scale-[1.02]"
   313→                      : "hover:border-primary/50"
   314→                  }`}
   315→                  onDragEnter={handleDrag}
   316→                  onDragLeave={handleDrag}
   317→                  onDragOver={handleDrag}
   318→                  onDrop={handleDrop}
   319→                >
   320→                  <input
   321→                    type="file"
   322→                    id="photo-upload"
   323→                    multiple
   324→                    accept="image/*"
   325→                    className="hidden"
   326→                    ref={fileInputRef}
   327→                    onChange={(e) => e.target.files && handleUpload(e.target.files)}
   328→                  />
   329→                  <label htmlFor="photo-upload" className="cursor-pointer">
   330→                    <motion.div
   331→                      animate={dragActive ? { scale: 1.1 } : { scale: 1 }}
   332→                    >
   333→                      <ImagePlus className="h-12 w-12 mx-auto text-muted-foreground mb-3" />
   334→                    </motion.div>
   335→                    <p className="font-medium">
   336→                      {dragActive
   337→                        ? "Suelta las fotos aquí"
   338→                        : "Arrastra fotos aquí o haz clic para subir"}
   339→                    </p>
   340→                    <p className="text-sm text-muted-foreground mt-1">
   341→                      Máximo 5 fotos · Cualquier formato de imagen
   342→                    </p>
   343→                  </label>
   344→                </div>
   345→
   346→                {/* Upload progress */}
   347→                {uploading && (
   348→                  <motion.div
   349→                    initial={{ opacity: 0, height: 0 }}
   350→                    animate={{ opacity: 1, height: "auto" }}
   351→                    className="space-y-2"
   352→                  >
   353→                    <div className="flex items-center gap-2">
   354→                      <Loader2 className="h-4 w-4 animate-spin text-primary" />
   355→                      <p className="text-sm text-primary font-medium">
   356→                        Subiendo imágenes...
   357→                      </p>
   358→                    </div>
   359→                    <Progress value={uploadProgress} className="h-1.5" />
   360→                  </motion.div>
   361→                )}
   362→
   363→                {errors.images && (
   364→                  <p className="text-xs text-destructive">{errors.images}</p>
   365→                )}
   366→
   367→                {/* Image previews */}
   368→                {form.images.length > 0 && (
   369→                  <div className="grid grid-cols-5 gap-2">
   370→                    <AnimatePresence>
   371→                      {form.images.map((img, i) => (
   372→                        <motion.div
   373→                          key={img + i}
   374→                          initial={{ opacity: 0, scale: 0.8 }}
   375→                          animate={{ opacity: 1, scale: 1 }}
   376→                          exit={{ opacity: 0, scale: 0.8 }}
   377→                          className="relative group"
   378→                        >
   379→                          <img
   380→                            src={img}
   381→                            alt=""
   382→                            className="w-full h-20 object-cover rounded-lg border"
   383→                          />
   384→                          {i === 0 && (
   385→                            <Badge className="absolute bottom-1 left-1 text-[8px] px-1 py-0 bg-primary">
   386→                              Portada
   387→                            </Badge>
   388→                          )}
   389→                          <button
   390→                            onClick={() => removeImage(i)}
   391→                            className="absolute -top-1.5 -right-1.5 bg-destructive text-white rounded-full w-5 h-5 flex items-center justify-center text-xs opacity-0 group-hover:opacity-100 transition-opacity shadow-sm"
   392→                          >
   393→                            <X className="h-3 w-3" />
   394→                          </button>
   395→                        </motion.div>
   396→                      ))}
   397→                    </AnimatePresence>
   398→                  </div>
   399→                )}
   400→
   401→                <div className="space-y-2">
   402→                  <Label>Video (URL opcional)</Label>
   403→                  <Input
   404→                    type="url"
   405→                    placeholder="https://youtube.com/..."
   406→                    value={form.videoUrl}
   407→                    onChange={(e) =>
   408→                      setForm((f) => ({ ...f, videoUrl: e.target.value }))
   409→                    }
   410→                  />
   411→                </div>
   412→              </motion.div>
   413→            )}
   414→
   415→            {/* Step 2: Details */}
   416→            {step === 2 && (
   417→              <motion.div
   418→                key="step2"
   419→                initial={{ opacity: 0, x: 20 }}
   420→                animate={{ opacity: 1, x: 0 }}
   421→                exit={{ opacity: 0, x: -20 }}
   422→                transition={{ duration: 0.2 }}
   423→                className="space-y-4"
   424→              >
   425→                <div className="space-y-2">
   426→                  <Label>Título *</Label>
   427→                  <Input
   428→                    placeholder="Ej: Cemento Portland 50kg"
   429→                    value={form.title}
   430→                    onChange={(e) =>
   431→                      setForm((f) => ({ ...f, title: e.target.value }))
   432→                    }
   433→                    className={errors.title ? "border-destructive" : ""}
   434→                  />
   435→                  {errors.title && (
   436→                    <p className="text-xs text-destructive">{errors.title}</p>
   437→                  )}
   438→                </div>
   439→
   440→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   441→                  <div className="space-y-2">
   442→                    <Label>Precio (C$) *</Label>
   443→                    <div className="relative">
   444→                      <DollarSign className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   445→                      <Input
   446→                        type="number"
   447→                        placeholder="0.00"
   448→                        value={form.price}
   449→                        onChange={(e) =>
   450→                          setForm((f) => ({ ...f, price: e.target.value }))
   451→                        }
   452→                        className={`pl-9 ${errors.price ? "border-destructive" : ""}`}
   453→                      />
   454→                    </div>
   455→                    {errors.price && (
   456→                      <p className="text-xs text-destructive">{errors.price}</p>
   457→                    )}
   458→                  </div>
   459→                  <div className="space-y-2">
   460→                    <Label>Categoría *</Label>
   461→                    <Select
   462→                      value={form.category}
   463→                      onValueChange={(v) =>
   464→                        setForm((f) => ({ ...f, category: v }))
   465→                      }
   466→                    >
   467→                      <SelectTrigger
   468→                        className={errors.category ? "border-destructive" : ""}
   469→                      >
   470→                        <SelectValue placeholder="Seleccionar" />
   471→                      </SelectTrigger>
   472→                      <SelectContent>
   473→                        {PRODUCT_CATEGORIES.map((c) => (
   474→                          <SelectItem key={c} value={c}>
   475→                            {c}
   476→                          </SelectItem>
   477→                        ))}
   478→                      </SelectContent>
   479→                    </Select>
   480→                    {errors.category && (
   481→                      <p className="text-xs text-destructive">
   482→                        {errors.category}
   483→                      </p>
   484→                    )}
   485→                  </div>
   486→                </div>
   487→
   488→                <div className="space-y-2">
   489→                  <Label>Descripción *</Label>
   490→                  <Textarea
   491→                    placeholder="Describe tu producto en detalle..."
   492→                    rows={4}
   493→                    value={form.description}
   494→                    onChange={(e) =>
   495→                      setForm((f) => ({ ...f, description: e.target.value }))
   496→                    }
   497→                    className={errors.description ? "border-destructive" : ""}
   498→                  />
   499→                  <div className="flex justify-between">
   500→                    {errors.description ? (
   501→                      <p className="text-xs text-destructive">
   502→                        {errors.description}
   503→                      </p>
   504→                    ) : (
   505→                      <span />
   506→                    )}
   507→                    <p className="text-xs text-muted-foreground">
   508→                      {form.description.length}/500
   509→                    </p>
   510→                  </div>
   511→                </div>
   512→
   513→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   514→                  <div className="space-y-2">
   515→                    <Label>Cantidad disponible</Label>
   516→                    <Input
   517→                      type="number"
   518→                      value={form.quantity}
   519→                      onChange={(e) =>
   520→                        setForm((f) => ({ ...f, quantity: e.target.value }))
   521→                      }
   522→                      min="1"
   523→                    />
   524→                  </div>
   525→                  <div className="space-y-2">
   526→                    <Label>Etiquetas</Label>
   527→                    <Input
   528→                      placeholder="construccion, cemento"
   529→                      value={form.tags}
   530→                      onChange={(e) =>
   531→                        setForm((f) => ({ ...f, tags: e.target.value }))
   532→                      }
   533→                    />
   534→                  </div>
   535→                </div>
   536→              </motion.div>
   537→            )}
   538→
   539→            {/* Step 3: Discount */}
   540→            {step === 3 && (
   541→              <motion.div
   542→                key="step3"
   543→                initial={{ opacity: 0, x: 20 }}
   544→                animate={{ opacity: 1, x: 0 }}
   545→                exit={{ opacity: 0, x: -20 }}
   546→                transition={{ duration: 0.2 }}
   547→                className="space-y-4"
   548→              >
   549→                <div className="flex items-center gap-2 mb-4">
   550→                  <Tag className="h-5 w-5 text-dorado" />
   551→                  <p className="text-sm text-muted-foreground">
   552→                    Opcional: Aplica un descuento a tu producto
   553→                  </p>
   554→                </div>
   555→
   556→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   557→                  <div className="space-y-2">
   558→                    <Label>Precio con descuento (C$)</Label>
   559→                    <Input
   560→                      type="number"
   561→                      placeholder="0.00"
   562→                      value={form.discountPrice}
   563→                      onChange={(e) => {
   564→                        const dp = e.target.value
   565→                        const priceNum = parseFloat(form.price)
   566→                        const pct =
   567→                          dp && priceNum > 0
   568→                            ? String(
   569→                                Math.round(
   570→                                  (1 - parseFloat(dp) / priceNum) *
   571→                                    100
   572→                                )
   573→                              )
   574→                            : ""
   575→                        setForm((f) => ({
   576→                          ...f,
   577→                          discountPrice: dp,
   578→                          discountPercent: pct,
   579→                        }))
   580→                      }}
   581→                    />
   582→                  </div>
   583→                  <div className="space-y-2">
   584→                    <Label>Descuento (%)</Label>
   585→                    <Input
   586→                      type="number"
   587→                      placeholder="0"
   588→                      value={form.discountPercent}
   589→                      onChange={(e) => {
   590→                        const pct = e.target.value
   591→                        const priceNum = parseFloat(form.price)
   592→                        const dp =
   593→                          pct && priceNum > 0
   594→                            ? String(
   595→                                priceNum *
   596→                                  (1 - parseFloat(pct) / 100)
   597→                              )
   598→                            : ""
   599→                        setForm((f) => ({
   600→                          ...f,
   601→                          discountPercent: pct,
   602→                          discountPrice: dp,
   603→                        }))
   604→                      }}
   605→                    />
   606→                  </div>
   607→                </div>
   608→
   609→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   610→                  <div className="space-y-2">
   611→                    <Label>Inicio del descuento</Label>
   612→                    <Input
   613→                      type="date"
   614→                      value={form.discountStart}
   615→                      onChange={(e) =>
